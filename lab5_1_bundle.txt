#include "lab5_1Acquisition.h"
#include "lab5_1Shared.h"
#include "../edDigitalThermometer/edDigitalThermometer.h"
#include "../edPotentiometer/edPotentiometer.h"

void lab5_1AcquisitionInit()
{
    edDigitalThermometerSetup();
    edPotentiometerSetup();
}

void vTask5_1Acquisition(void *pvParameters)
{
    (void)pvParameters;

    TickType_t xLastWakeTime = xTaskGetTickCount();
    const TickType_t xPeriod = pdMS_TO_TICKS(LAB5_1_TASK_ACQ_MS);

    for (;;) {
        edDigitalThermometerLoop();
        edPotentiometerLoop();

        const float latestTempC = edDigitalThermometerGetTemperatureC();
        const int latestPotAdc = edPotentiometerGetRawAdc();

        if ((gLab5_1Mutex != NULL) && (xSemaphoreTake(gLab5_1Mutex, pdMS_TO_TICKS(10)) == pdTRUE)) {
            gLab5_1State.tempRawC = latestTempC;
            gLab5_1State.potRawAdc = latestPotAdc;
            gLab5_1State.sampleCounter++;
            xSemaphoreGive(gLab5_1Mutex);
        }

        vTaskDelayUntil(&xLastWakeTime, xPeriod);
    }
}

#include "lab5_1Conditioning.h"
#include "lab5_1Shared.h"

static float mapPotToSetPointC(int potAdc)
{
    const float normalized = (float)potAdc / 1023.0f;
    const float span = LAB5_1_SETPOINT_MAX_C - LAB5_1_SETPOINT_MIN_C;
    return LAB5_1_SETPOINT_MIN_C + (normalized * span);
}

void vTask5_1Conditioning(void *pvParameters)
{
    (void)pvParameters;

    TickType_t xLastWakeTime = xTaskGetTickCount();
    const TickType_t xPeriod = pdMS_TO_TICKS(LAB5_1_TASK_COND_MS);

    float tempWindow[4] = {20.0f, 20.0f, 20.0f, 20.0f};
    uint8_t windowPos = 0;

    for (;;) {
        float rawTempC = 20.0f;
        int potRawAdc = 0;

        if ((gLab5_1Mutex != NULL) && (xSemaphoreTake(gLab5_1Mutex, pdMS_TO_TICKS(10)) == pdTRUE)) {
            rawTempC = gLab5_1State.tempRawC;
            potRawAdc = gLab5_1State.potRawAdc;
            xSemaphoreGive(gLab5_1Mutex);
        }

        const float boundedSetPointC = lab5_1ClampFloat(
            mapPotToSetPointC(potRawAdc),
            LAB5_1_SETPOINT_MIN_C,
            LAB5_1_SETPOINT_MAX_C
        );

        tempWindow[windowPos] = rawTempC;
        windowPos = (windowPos + 1) % 4;

        float tempAverage = 0.0f;
        for (uint8_t i = 0; i < 4; ++i) {
            tempAverage += tempWindow[i];
        }
        tempAverage *= 0.25f;

        const float hyst = LAB5_1_DEFAULT_HYST_C;
        const float lower = boundedSetPointC - 0.5f * hyst;
        const float upper = boundedSetPointC + 0.5f * hyst;

        if ((gLab5_1Mutex != NULL) && (xSemaphoreTake(gLab5_1Mutex, pdMS_TO_TICKS(10)) == pdTRUE)) {
            gLab5_1State.setPointC = boundedSetPointC;
            gLab5_1State.hysteresisC = hyst;
            gLab5_1State.tempFilteredC = tempAverage;
            gLab5_1State.lowerThresholdC = lower;
            gLab5_1State.upperThresholdC = upper;
            xSemaphoreGive(gLab5_1Mutex);
        }

        vTaskDelayUntil(&xLastWakeTime, xPeriod);
    }
}

#include "lab5_1Control.h"
#include "lab5_1Shared.h"
#include "../edRelay/edRelay.h"

void vTask5_1Control(void *pvParameters)
{
    (void)pvParameters;

    TickType_t xLastWakeTime = xTaskGetTickCount();
    const TickType_t xPeriod = pdMS_TO_TICKS(LAB5_1_TASK_CTRL_MS);

    bool relayOn = false;

    for (;;) {
        float measuredTempC = 20.0f;
        float lowerThresholdC = 24.5f;
        float upperThresholdC = 25.5f;

        if ((gLab5_1Mutex != NULL) && (xSemaphoreTake(gLab5_1Mutex, pdMS_TO_TICKS(10)) == pdTRUE)) {
            measuredTempC = gLab5_1State.tempFilteredC;
            lowerThresholdC = gLab5_1State.lowerThresholdC;
            upperThresholdC = gLab5_1State.upperThresholdC;
            relayOn = gLab5_1State.relayStateOn;
            xSemaphoreGive(gLab5_1Mutex);
        }

        if (!relayOn) {
            if (measuredTempC < lowerThresholdC) {
                relayOn = true;
            }
        } else {
            if (measuredTempC > upperThresholdC) {
                relayOn = false;
            }
        }

        if (relayOn) {
            edRelayOn();
        } else {
            edRelayOff();
        }

        if ((gLab5_1Mutex != NULL) && (xSemaphoreTake(gLab5_1Mutex, pdMS_TO_TICKS(10)) == pdTRUE)) {
            gLab5_1State.relayStateOn = relayOn;
            xSemaphoreGive(gLab5_1Mutex);
        }

        vTaskDelayUntil(&xLastWakeTime, xPeriod);
    }
}

#include "lab5_1Report.h"
#include "lab5_1Shared.h"
#include "../ddLcd/ddLcd.h"

#include <stdio.h>

static void splitCelsius1(float valueC, int *integerPart, int *frac1)
{
    int scaled = (int)(valueC * 10.0f);
    *integerPart = scaled / 10;
    *frac1 = scaled % 10;
    if (*frac1 < 0) {
        *frac1 = -*frac1;
    }
}

void vTask5_1Report(void *pvParameters)
{
    (void)pvParameters;

    TickType_t xLastWakeTime = xTaskGetTickCount();
    const TickType_t xPeriod = pdMS_TO_TICKS(LAB5_1_TASK_REP_MS);

    for (;;) {
        vTaskDelayUntil(&xLastWakeTime, xPeriod);

        lab5_1State_t snapshot;
        if ((gLab5_1Mutex == NULL) || (xSemaphoreTake(gLab5_1Mutex, pdMS_TO_TICKS(20)) != pdTRUE)) {
            continue;
        }
        snapshot = gLab5_1State;
        xSemaphoreGive(gLab5_1Mutex);

        int tempI = 0;
        int tempF = 0;
        int setI = 0;
        int setF = 0;
        int hystI = 0;
        int hystF = 0;
        int lowI = 0;
        int lowF = 0;
        int upI = 0;
        int upF = 0;

        splitCelsius1(snapshot.tempFilteredC, &tempI, &tempF);
        splitCelsius1(snapshot.setPointC, &setI, &setF);
        splitCelsius1(snapshot.hysteresisC, &hystI, &hystF);
        splitCelsius1(snapshot.lowerThresholdC, &lowI, &lowF);
        splitCelsius1(snapshot.upperThresholdC, &upI, &upF);

        char line0[17];
        char line1[17];

        snprintf(line0, sizeof(line0), "T:%2d.%1d SP:%2d.%1d", tempI, tempF, setI, setF);
        snprintf(line1, sizeof(line1), "R:%s H:%1d.%1d", snapshot.relayStateOn ? "ON " : "OFF", hystI, hystF);

        ddLcdClear();
        ddLcdSetCursor(0, 0);
        ddLcdPrint(line0);
        ddLcdSetCursor(0, 1);
        ddLcdPrint(line1);

        printf(
            "L5.1 temp=%d.%dC set=%d.%dC hyst=%d.%dC low=%d.%dC high=%d.%dC relay=%s adc=%d samples=%lu\r\n",
            tempI,
            tempF,
            setI,
            setF,
            hystI,
            hystF,
            lowI,
            lowF,
            upI,
            upF,
            snapshot.relayStateOn ? "ON" : "OFF",
            snapshot.potRawAdc,
            (unsigned long)snapshot.sampleCounter
        );
    }
}

===== src\lab5_1\lab5_1Shared.cpp =====

#include "lab5_1Shared.h"

lab5_1State_t gLab5_1State;
SemaphoreHandle_t gLab5_1Mutex = NULL;

void lab5_1SharedInit()
{
    gLab5_1State.tempRawC = 20.0f;
    gLab5_1State.tempFilteredC = 20.0f;
    gLab5_1State.potRawAdc = 0;
    gLab5_1State.setPointC = 25.0f;
    gLab5_1State.hysteresisC = LAB5_1_DEFAULT_HYST_C;
    gLab5_1State.lowerThresholdC = gLab5_1State.setPointC - 0.5f * gLab5_1State.hysteresisC;
    gLab5_1State.upperThresholdC = gLab5_1State.setPointC + 0.5f * gLab5_1State.hysteresisC;
    gLab5_1State.relayStateOn = false;
    gLab5_1State.sampleCounter = 0;

    gLab5_1Mutex = xSemaphoreCreateMutex();
}

float lab5_1ClampFloat(float value, float minValue, float maxValue)
{
    if (value < minValue) {
        return minValue;
    }
    if (value > maxValue) {
        return maxValue;
    }
    return value;
}
